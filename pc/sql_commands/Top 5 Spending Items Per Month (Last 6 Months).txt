USE [excel_imports]
GO

WITH RankedIndividualExpenses AS (
    SELECT
        [Date],
        YEAR([Date]) AS ExpenseYear,
        MONTH([Date]) AS ExpenseMonthNumber,
        DATENAME(month, [Date]) AS ExpenseMonthName,
        [Description],
        [Category],
        [Your Expense],
        ROW_NUMBER() OVER (
            PARTITION BY YEAR([Date]), MONTH([Date])
            ORDER BY [Your Expense] DESC
        ) AS RowRank
    FROM
        [dbo].[All_My_Expenses_Combined] -- Or [dbo].[All_My_Expenses_Combined_New] if you haven't renamed it yet
    WHERE
        [Date] IS NOT NULL
        AND [Your Expense] IS NOT NULL
        -- Filter for data within the last 6 months from today's date
        AND [Date] >= DATEADD(month, -6, GETDATE())
)
SELECT
    ExpenseYear,
    ExpenseMonthName,
    [Date],
    [Description],
    [Category],
    [Your Expense]
FROM
    RankedIndividualExpenses
WHERE
    RowRank <= 5
ORDER BY
    ExpenseYear DESC,
    ExpenseMonthNumber DESC,
    [Your Expense] DESC;
GO
