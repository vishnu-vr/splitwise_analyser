USE [excel_imports]
GO

WITH MonthlyDescriptionSpending AS (
    SELECT
        YEAR([Date]) AS ExpenseYear,
        MONTH([Date]) AS ExpenseMonthNumber,
        DATENAME(month, [Date]) AS ExpenseMonthName,
        [Description],
        [Category], -- Added Category
        SUM([Your Expense]) AS MonthlyDescriptionTotal
    FROM
        [dbo].[All_My_Expenses_Combined] -- Or [dbo].[All_My_Expenses_Combined_New] if you haven't renamed it yet
    WHERE
        [Date] IS NOT NULL
        AND [Your Expense] IS NOT NULL
        -- Filter for data within the last 6 months from today's date
        AND [Date] >= DATEADD(month, -6, GETDATE())
    GROUP BY
        YEAR([Date]),
        MONTH([Date]),
        DATENAME(month, [Date]),
        [Description],
        [Category] -- Added Category to GROUP BY
),
RankedSpending AS (
    SELECT
        ExpenseYear,
        ExpenseMonthNumber,
        ExpenseMonthName,
        [Description],
        [Category], -- Added Category
        MonthlyDescriptionTotal,
        ROW_NUMBER() OVER (PARTITION BY ExpenseYear, ExpenseMonthNumber ORDER BY MonthlyDescriptionTotal DESC) AS RowRank
    FROM
        MonthlyDescriptionSpending
)
SELECT
    ExpenseYear,
    ExpenseMonthName,
    [Description],
    [Category], -- Added Category
    MonthlyDescriptionTotal
FROM
    RankedSpending
WHERE
    RowRank <= 5
ORDER BY
    ExpenseYear DESC,
    ExpenseMonthNumber DESC,
    MonthlyDescriptionTotal DESC;
GO
